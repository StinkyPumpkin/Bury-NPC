cmake_minimum_required(VERSION 3.21)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")
project(BuryTakeBodies VERSION 1.6.2 LANGUAGES CXX)

# Deploy target: the mod FOLDER gets --Claude; the DLL name stays plain.
set(OUTPUT_FOLDER "X:/MODDINGSSE/modorganizer2/mods/BuryTakeBodies--Claude")

# 1.6.1: CommonLibSSE-NG from alandtse (github.com/alandtse/CommonLibSSE-NG), built from source out of the pinned
# submodule extern/CommonLibSSE-NG (tag v9.1.0). It reads the Address Library v5 database used by Skyrim 1.7.x and
# sets the AddressLibraryV5 flag, so 1.7.x SKSE loads the DLL. Up to 1.6.0 the vcpkg commonlibsse-ng 3.7.0 port was
# used: it classifies 1.7.x as SE and stops with "failed to open address library file".
set(CLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/extern/CommonLibSSE-NG")
if(NOT EXISTS "${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")
    message(FATAL_ERROR "extern/CommonLibSSE-NG is empty. Run: git submodule update --init --recursive --depth 1")
endif()
if(NOT EXISTS "${CLIB_SOURCE_DIR}/extern/openvr/lib/win64/openvr_api.lib")
    message(FATAL_ERROR "extern/CommonLibSSE-NG/extern/openvr is empty (VR is enabled). Run: git submodule update --init --recursive --depth 1")
endif()

# Every CACHE FORCE must precede add_subdirectory: option() inside NG cannot override an existing cache entry.
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)                 # NG defaults ON -> find_package(Catch2 REQUIRED)
set(SKSE_SUPPORT_XBYAK OFF CACHE BOOL "" FORCE)          # no trampoline hooks
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)   # no write_branch sites; ON downloads minhook at configure time
# One DLL for every runtime, dispatched at run time.
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)             # AE = 1.6.x AND 1.7.x
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
set(REX_OPTION_INI OFF CACHE BOOL "" FORCE)
set(REX_OPTION_JSON OFF CACHE BOOL "" FORCE)
set(REX_OPTION_TOML OFF CACHE BOOL "" FORCE)

add_subdirectory("${CLIB_SOURCE_DIR}" CommonLibSSE-NG)
file(READ "${CLIB_SOURCE_DIR}/vcpkg.json" _clib_manifest)
string(JSON BTB_CLIB_VERSION GET "${_clib_manifest}" "version-semver")
message(STATUS "CommonLibSSE-NG ${BTB_CLIB_VERSION} from ${CLIB_SOURCE_DIR}")
# NG's source build does not include its plugin helper (only the prebuilt path does).
include("${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")

# Declaration generated by add_commonlibsse_plugin (SKSEPlugin_Version + SKSEPlugin_Query):
#   versionIndependence = AddressLibraryPostAE, versionIndependenceEx = AddressLibraryV5 | NoStructUse.
# MINIMUM_SKSE_VERSION must be a full x.y.z triplet (the helper's bare "0" default does not compile).
add_commonlibsse_plugin(${PROJECT_NAME} MINIMUM_SKSE_VERSION 0.0.0 SOURCES
    src/main.cpp
    src/Settings.cpp
    src/PromptManager.cpp
    src/RespectManager.cpp
    src/PickupManager.cpp
    src/MsvcShim.cpp
)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_23)
target_precompile_headers(${PROJECT_NAME} PRIVATE src/pch.h)
target_compile_definitions(${PROJECT_NAME} PRIVATE BTB_COMMONLIB_VERSION="${BTB_CLIB_VERSION}")

# SimpleIni, SkyPrompt API and QuickLoot API headers are vendored in include/.
target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/include
)

set(DLL_FOLDER "${OUTPUT_FOLDER}/SKSE/Plugins")
message(STATUS "SKSE plugin output: ${DLL_FOLDER}")

add_custom_command(
    TARGET "${PROJECT_NAME}" POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${DLL_FOLDER}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:${PROJECT_NAME}>" "${DLL_FOLDER}/$<TARGET_FILE_NAME:${PROJECT_NAME}>"
    VERBATIM
)
